<script>

/* PLAY */

function openGame(game){

let plays =
Number(localStorage.getItem(game)||0);

localStorage.setItem(
game,
plays+1
);

alert("Launching " + game);

}

/* SEARCH */

document.getElementById("search")
.addEventListener("keyup",function(){

let value =
this.value.toLowerCase();

document
.querySelectorAll(".card")
.forEach(card=>{

let text =
card.querySelector("h3")
.textContent
.toLowerCase();

card.style.display=
text.includes(value)
?"block"
:"none";

});

});

/* CATEGORY */

document.getElementById("category")
.addEventListener("change",function(){

let cat=this.value;

document
.querySelectorAll(".card")
.forEach(card=>{

card.style.display=
cat==="all"||
card.dataset.category===cat
?"block"
:"none";

});

});

/* HOME */

function showAllGames(){

document.getElementById("search").value="";
document.getElementById("category").value="all";

document
.querySelectorAll(".card")
.forEach(card=>{

card.style.display="block";

});

}

/* WISHLIST */

function toggleFavorite(game){

let favs=
JSON.parse(
localStorage.getItem("favorites")
||"[]"
);

if(favs.includes(game)){

favs=
favs.filter(
item=>item!==game
);

}else{

favs.push(game);

}

localStorage.setItem(
"favorites",
JSON.stringify(favs)
);

updateButtons();

}

function updateButtons(){

let favs=
JSON.parse(
localStorage.getItem("favorites")
||"[]"
);

document
.querySelectorAll(".fav-btn")
.forEach(btn=>{

let game=
btn.parentElement
.querySelector("h3")
.textContent;

btn.textContent=
favs.includes(game)
?"Remove Wishlist"
:"Add Wishlist";

});

}

function showFavorites(){

let favs=
JSON.parse(
localStorage.getItem("favorites")
||"[]"
);

document
.querySelectorAll(".card")
.forEach(card=>{

let game=
card.querySelector("h3")
.textContent;

card.style.display=
favs.includes(game)
?"block"
:"none";

});

}

/* TRENDING */

function showTrending(){

let found=false;

document
.querySelectorAll(".card")
.forEach(card=>{

let game=
card.querySelector("h3")
.textContent;

let plays=
Number(
localStorage.getItem(game)||0
);

if(plays>0){

card.style.display="block";
found=true;

}else{

card.style.display="none";

}

});

if(!found){

alert(
"Play some games first to create trending games."
);

}

}

updateButtons();

</script>
